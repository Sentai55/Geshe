local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local LocalPlayer = Players.LocalPlayer

-- SAKLAR UTAMA (Setel true untuk jalan, setel false untuk berhenti)
_G.AutoFarmStory = true

--------------------------------------------------------------------------------
-- PENGATURAN KORDINAT CHECKPOINT & OFFSET (MODE BIASA)
--------------------------------------------------------------------------------
local CHECKPOINTS = {
    Vector3.new(-6141, -202, 587),  -- Kordinat 1
    Vector3.new(-4090, 441, 1572),  -- Kordinat 2
    Vector3.new(-4292, 446, 3175),  -- Kordinat 3
    Vector3.new(-8871, 199, -1651), -- Kordinat 4
    Vector3.new(-4373, 521, -1947), -- Kordinat 5 (Terakhir)
}

local SAFE_OFFSET_Y = 0       -- 0 Karena kordinat kamu sudah di bawah tanah
local STUCK_OFFSET_Y = -8     -- Offset -8 stud di bawah NPC jika nyangkut
local NO_DAMAGE_TIMEOUT = 30  -- Batas waktu (detik) jika NPC nyangkut

--------------------------------------------------------------------------------
-- VARIABEL STATUS
--------------------------------------------------------------------------------
local currentStageIndex = 1
local lastDamageTime = tick()
local isFinished = false

--------------------------------------------------------------------------------
-- PENDETEKSI REMOTE DAMAGE (DmgEffectURE)
--------------------------------------------------------------------------------
task.spawn(function()
    local remotes = ReplicatedStorage:WaitForChild("Remotes", 15)
    if remotes then
        local dmgRemote = remotes:WaitForChild("DmgEffectURE", 15)
        if dmgRemote and dmgRemote:IsA("RemoteEvent") then
            dmgRemote.OnClientEvent:Connect(function(...)
                lastDamageTime = tick()
            end)
        end
    end
end)

--------------------------------------------------------------------------------
-- LOGIKA UTAMA AUTO FARM MODE BIASA
--------------------------------------------------------------------------------
task.spawn(function()
    -- Memastikan EnemyFolder terdeteksi walau game baru loading di AutoExec
    local EnemyFolder = workspace:WaitForChild("EnemyNpc", 30)
    
    if not EnemyFolder then
        warn("Folder 'EnemyNpc' belum ter-load!")
        return
    end

    local function getAliveNPCs()
        local aliveList = {}
        for _, npc in ipairs(EnemyFolder:GetChildren()) do
            local npcHrp = npc:FindFirstChild("HumanoidRootPart") or npc.PrimaryPart
            local npcHumanoid = npc:FindFirstChildOfClass("Humanoid")

            local isAlive = npcHrp and (not npcHumanoid or (npcHumanoid.Health > 0 and npcHumanoid:GetState() ~= Enum.HumanoidStateType.Dead))
            if isAlive then
                table.insert(aliveList, npc)
            end
        end
        return aliveList
    end

    print("🚀 Auto Farm Mode Biasa Dimulai dari Kordinat 1!")

    while _G.AutoFarmStory and task.wait(0.05) do
        pcall(function()
            if isFinished then return end

            local character = LocalPlayer.Character
            if not character or not character:FindFirstChild("HumanoidRootPart") then return end
            
            local hrp = character.HumanoidRootPart
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            local aliveNPCs = getAliveNPCs()

            -- Penahan Gravitasi Sumbu Y
            local freezeVel = hrp:FindFirstChild("PositionLock")
            if not freezeVel then
                freezeVel = Instance.new("BodyVelocity")
                freezeVel.Name = "PositionLock"
                freezeVel.MaxForce = Vector3.new(0, math.huge, 0)
                freezeVel.Velocity = Vector3.zero
                freezeVel.Parent = hrp
            end

            -- Jika masih ada NPC hidup
            if #aliveNPCs > 0 then
                if humanoid then
                    humanoid.AutoRotate = false
                    humanoid:ChangeState(Enum.HumanoidStateType.Physics)
                end

                local currentPoint = CHECKPOINTS[currentStageIndex]
                local targetCFrame = nil
                
                local timeSinceLastDamage = tick() - lastDamageTime

                -- BILA 30 DETIK GAK ADA DAMAGE (Berlaku berapa pun jumlah NPC yang tersisa)
                if timeSinceLastDamage >= NO_DAMAGE_TIMEOUT then
                    -- Samperin NPC pertama yang terdeteksi di daftar musuh hidup
                    local targetNpc = aliveNPCs[1]
                    local targetHrp = targetNpc:FindFirstChild("HumanoidRootPart") or targetNpc.PrimaryPart
                    
                    if targetHrp then
                        targetCFrame = targetHrp.CFrame * CFrame.new(0, STUCK_OFFSET_Y, 0)
                    end
                else
                    -- Jika damage masih berjalan lancar, tetap diam di kordinat aktif
                    targetCFrame = CFrame.new(currentPoint) * CFrame.new(0, SAFE_OFFSET_Y, 0)
                end

                -- Kunci Posisi & Rotasi Miring 90 Derajat ke Atas
                if targetCFrame then
                    hrp.CFrame = targetCFrame * CFrame.Angles(math.rad(90), 0, 0)
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                end

            else
                -- BILA NPC KOSONG / HABIS
                lastDamageTime = tick()

                if currentStageIndex < #CHECKPOINTS then
                    -- Pindah ke Kordinat Selanjutnya
                    currentStageIndex = currentStageIndex + 1
                    print("Ea NPC Habis! Pindah ke Kordinat : " .. currentStageIndex)
                    task.wait(0.5) -- Jeda sebentar sebelum kunci kordinat baru
                else
                    -- KORDINAT TERAKHIR SELESAI
                    if not isFinished then
                        isFinished = true
                        print("🎉 Semua Kordinat Selesai! Menunggu 3 Detik...")

                        -- Kembalikan posisi berdiri normal
                        if humanoid then
                            humanoid.AutoRotate = true
                            humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                        end
                        if freezeVel then
                            freezeVel:Destroy()
                        end

                        -- Tunggu 3 Detik
                        task.wait(3)

                        -- Fire Remote Play Again
                        local gameRoundRemote = ReplicatedStorage:WaitForChild("Remotes", 10):WaitForChild("GameRoundRE", 10)
                        if gameRoundRemote then
                            gameRoundRemote:FireServer("VotePlayAgain")
                            print("🔄 Remote VotePlayAgain Berhasil Ditembak!")
                        end
                    end
                end
            end
        end)
    end

    -- Pembersihan otomatis jika _G.AutoFarmStory = false
    local character = LocalPlayer.Character
    if character then
        local hrp = character:FindFirstChild("HumanoidRootPart")
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.AutoRotate = true
            humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
        if hrp and hrp:FindFirstChild("PositionLock") then
            hrp.PositionLock:Destroy()
        end
    end
end)
