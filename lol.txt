local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local LocalPlayer = Players.LocalPlayer

-- Pengaturan Ketinggian & Timer
local SAFE_OFFSET_Y = -10
local STUCK_OFFSET_Y = -7
local NO_DAMAGE_TIMEOUT = 30

-- Variabel Status
local initialCFrame = nil
local lastSingleNpc = nil
local lastDamageTime = tick()

--------------------------------------------------------------------------------
-- PENDETEKSI REMOTE DAMAGE (DmgEffectURE)
--------------------------------------------------------------------------------
task.spawn(function()
    local remotes = ReplicatedStorage:WaitForChild("Remotes", 10)
    if remotes then
        local dmgRemote = remotes:WaitForChild("DmgEffectURE", 10)
        if dmgRemote and dmgRemote:IsA("RemoteEvent") then
            dmgRemote.OnClientEvent:Connect(function(...)
                lastDamageTime = tick()
            end)
        end
    end
end)

--------------------------------------------------------------------------------
-- SISTEM NOCLIP (Menembus Tembok/Lantai)
--------------------------------------------------------------------------------
RunService.Stepped:Connect(function()
    local character = LocalPlayer.Character
    if character then
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                part.CanCollide = false
            end
        end
    end
end)

--------------------------------------------------------------------------------
-- LOGIKA UTAMA AUTO FARM (FLY + MENGHADAP KE ATAS)
--------------------------------------------------------------------------------
task.spawn(function()
    local EnemyFolder = workspace:FindFirstChild("EnemyNpc") or workspace:WaitForChild("EnemyNpc", 10)
    
    if not EnemyFolder then
        warn("Folder 'EnemyNpc' tidak ditemukan di Workspace!")
        return
    end

    local function getAliveNPCs()
        local aliveList = {}
        for _, npc in ipairs(EnemyFolder:GetChildren()) do
            local npcHrp = npc:FindFirstChild("HumanoidRootPart") or npc.PrimaryPart
            local npcHumanoid = npc:FindFirstChildOfClass("Humanoid")

            local isAlive = npcHrp and (not npcHumanoid or (npcHumanoid.Health > 0 and npcHumanoid:GetState() ~= Enum.HumanoidStateType.Dead))
            if isAlive then
                table.insert(aliveList, npc)
            end
        end
        return aliveList
    end

    while task.wait(0.1) do
        pcall(function()
            local character = LocalPlayer.Character
            if not character or not character:FindFirstChild("HumanoidRootPart") then return end
            
            local hrp = character.HumanoidRootPart
            local aliveNPCs = getAliveNPCs()

            -- Memastikan karakter dalam kondisi 'Fly' (membekukan gravitasi)
            local bodyVel = hrp:FindFirstChild("AutoFarmFly")
            if not bodyVel then
                bodyVel = Instance.new("BodyVelocity")
                bodyVel.Name = "AutoFarmFly"
                bodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                bodyVel.Velocity = Vector3.zero
                bodyVel.Parent = hrp
            end

            if #aliveNPCs > 0 then
                if not initialCFrame then
                    initialCFrame = hrp.CFrame
                end

                local targetCFrame = nil

                if #aliveNPCs == 1 then
                    local targetNpc = aliveNPCs[1]
                    
                    if lastSingleNpc ~= targetNpc then
                        lastSingleNpc = targetNpc
                        lastDamageTime = tick()
                    end

                    local timeSinceLastDamage = tick() - lastDamageTime

                    if timeSinceLastDamage >= NO_DAMAGE_TIMEOUT then
                        local targetHrp = targetNpc:FindFirstChild("HumanoidRootPart") or targetNpc.PrimaryPart
                        if targetHrp then
                            targetCFrame = targetHrp.CFrame * CFrame.new(0, STUCK_OFFSET_Y, 0)
                        end
                    else
                        targetCFrame = initialCFrame * CFrame.new(0, SAFE_OFFSET_Y, 0)
                    end
                else
                    lastSingleNpc = nil
                    lastDamageTime = tick()
                    targetCFrame = initialCFrame * CFrame.new(0, SAFE_OFFSET_Y, 0)
                end

                -- Terapkan CFrame + Putar 90 Derajat ke Atas (Head Facing Up)
                if targetCFrame then
                    hrp.CFrame = targetCFrame * CFrame.Angles(math.rad(90), 0, 0)
                end
            else
                -- Jika NPC bersih, kembalikan ke tempat semula dan hapus efek Fly
                if initialCFrame then
                    hrp.CFrame = initialCFrame
                    initialCFrame = nil
                end
                
                if bodyVel then
                    bodyVel:Destroy()
                end
                
                lastSingleNpc = nil
                lastDamageTime = tick()
            end
        end)
    end
end)
