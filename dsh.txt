task.delay(10, function()

    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local RunService = game:GetService("RunService")
    local LocalPlayer = Players.LocalPlayer

    -- SAKLAR UTAMA
    _G.AutoFarmStory = true

    --------------------------------------------------------------------------------
    -- DATABASE KOORDINAT MAP (BERDASARKAN PLACE ID)
    --------------------------------------------------------------------------------
    local MAP_DATA = {
        -- MAP 4
        [80693195250724] = {
            NAME = "Map 4",
            CHECKPOINTS = {
                Vector3.new(-6141, -202, 587),  -- Room 1
                Vector3.new(-4090, 441, 1572),  -- Room 2
                Vector3.new(-4292, 446, 3175),  -- Room 3
                Vector3.new(-8871, 199, -1651), -- Room 4
                Vector3.new(-4373, 521, -1947), -- Room 5 (Terakhir)
            },
            PORTALS = {
                [2] = Vector3.new(-6420, -170, 585),   -- Portal setelah Room 1
                [3] = Vector3.new(-4336, 465, 1584),   -- Portal setelah Room 2
                [4] = Vector3.new(-4044, 468, 3152),   -- Portal setelah Room 3
                [5] = Vector3.new(-8644, 221, -1196),  -- Portal setelah Room 4
            }
        },

        -- MAP 3
        [112316840155266] = {
            NAME = "Map 3",
            CHECKPOINTS = {
                Vector3.new(710, 58, 577),   -- Room 1
                Vector3.new(2, -13, 782),    -- Room 2
                Vector3.new(261, -3, 267),   -- Room 3
                Vector3.new(753, 58, -48),   -- Room 4
                Vector3.new(-905, 9, 264),   -- Room 5
                Vector3.new(-425, 24, 260),  -- Room 6 (Terakhir)
            },
            PORTALS = {
                [2] = Vector3.new(550, 89, 594),   -- Portal setelah Room 1
                [3] = Vector3.new(328, -3, 815),   -- Portal setelah Room 2
                [4] = Vector3.new(440, 27, 286),   -- Portal setelah Room 3
                [5] = Vector3.new(866, 93, -32),   -- Portal setelah Room 4
                -- Room 5 ke 6 tidak ada portal
            }
        }
    }

    --------------------------------------------------------------------------------
    -- DETEKSI MAP AKTIF
    --------------------------------------------------------------------------------
    local currentMap = MAP_DATA[game.PlaceId]

    if not currentMap then
        warn("⚠️ Place ID (" .. tostring(game.PlaceId) .. ") Belum Terdaftar di Script!")
        return
    end

    local CHECKPOINTS = currentMap.CHECKPOINTS
    local PORTAL_COORDS = currentMap.PORTALS
    local SAFE_OFFSET_Y = 0

    --------------------------------------------------------------------------------
    -- VARIABEL STATUS
    --------------------------------------------------------------------------------
    local currentStageIndex = 1
    local isFinished = false
    local isTransitioning = false

    --------------------------------------------------------------------------------
    -- SISTEM NOCLIP OTOMATIS
    --------------------------------------------------------------------------------
    RunService.Stepped:Connect(function()
        if _G.AutoFarmStory then
            local character = LocalPlayer.Character
            if character then
                for _, part in ipairs(character:GetDescendants()) do
                    if part:IsA("BasePart") and part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
        end
    end)

    --------------------------------------------------------------------------------
    -- FUNGSI CEK HASIL GAME (TAMAT)
    --------------------------------------------------------------------------------
    local function checkGameFinished()
        local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
        if playerGui then
            local resultGui = playerGui:FindFirstChild("ResultGui")
            if resultGui then
                local screenSettlement = resultGui:FindFirstChild("ScreenSettlement")
                if screenSettlement and screenSettlement.Visible == true then
                    return true
                end
            end
        end
        return false
    end

    --------------------------------------------------------------------------------
    -- FUNGSI AMBIL OBJECT UI ROUND COMPLETED
    --------------------------------------------------------------------------------
    local function getRoundCompletedUI()
        local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
        if playerGui then
            local battleHUD = playerGui:FindFirstChild("BattleHUD")
            if battleHUD then
                local informFrame = battleHUD:FindFirstChild("InformFrame")
                if informFrame then
                    return informFrame:FindFirstChild("RoundCompleted")
                end
            end
        end
        return nil
    end

    --------------------------------------------------------------------------------
    -- LOGIKA UTAMA AUTO FARM STORY
    --------------------------------------------------------------------------------
    task.spawn(function()
        print("📢 Auto Farm Story Aktif di: " .. currentMap.NAME .. " (Place ID: " .. tostring(game.PlaceId) .. ")")

        while _G.AutoFarmStory and task.wait(0.05) do
            pcall(function()
                if isFinished then return end

                local char = LocalPlayer.Character
                if not char or not char:FindFirstChild("HumanoidRootPart") then return end
                
                local charHrp = char.HumanoidRootPart
                local humanoid = char:FindFirstChildOfClass("Humanoid")

                -- 1. CEK HASIL GAME (TAMAT)
                if checkGameFinished() then
                    isFinished = true
                    print("🏆 ScreenSettlement Visible! Game Selesai...")

                    if humanoid then humanoid.AutoRotate = true end
                    task.wait(3)

                    local gameRoundRemote = ReplicatedStorage:WaitForChild("Remotes", 10):WaitForChild("GameRoundRE", 10)
                    if gameRoundRemote then
                        gameRoundRemote:FireServer("VotePlayAgain")
                        print("✉️ VotePlayAgain Ditembak!")
                    end
                    return
                end

                -- 2. PENAHAN GRAVITASI PERMANEN
                local freezeVel = charHrp:FindFirstChild("PositionLock")
                if not freezeVel then
                    freezeVel = Instance.new("BodyVelocity")
                    freezeVel.Name = "PositionLock"
                    freezeVel.MaxForce = Vector3.new(0, math.huge, 0)
                    freezeVel.Velocity = Vector3.zero
                    freezeVel.Parent = charHrp
                end

                -- 3. CEK STATUS UI
                local roundUI = getRoundCompletedUI()
                local isUIVisible = roundUI and roundUI.Visible == true

                -- LOGIKA UTAMA: PULL TRIGGER PAS VISIBLE = TRUE
                if isUIVisible and not isTransitioning then
                    
                    -- Kunci transisi langsung di awal biar loop tidak spam
                    isTransitioning = true
                    print("⚔️ RoundCompleted Visible! Menunggu 3 detik sebelum TP...")

                    -- WAJIB NUNGGU 3 DETIK SETELAH VISIBLE == TRUE
                    task.wait(3)

                    if currentStageIndex < #CHECKPOINTS then
                        currentStageIndex = currentStageIndex + 1

                        -- LEPAS LOCK GRAVITASI SEBELUM TP
                        local currentLock = charHrp:FindFirstChild("PositionLock")
                        if currentLock then currentLock:Destroy() end

                        -- TELEPORT KE PORTAL (JIKA ADA PORTAL)
                        local portalPos = PORTAL_COORDS[currentStageIndex]
                        if portalPos then
                            print("🌀 Transit di Portal " .. (currentStageIndex - 1))
                            charHrp.CFrame = CFrame.new(portalPos)
                            
                            -- Nunggu di portal buat trigger game
                            task.wait(3)
                        else
                            print("⚡ Tidak ada portal untuk Room " .. currentStageIndex .. " (Direct TP!)")
                        end

                        -- TELEPORT KE ROOM BANTAI SELANJUTNYA
                        local nextPoint = CHECKPOINTS[currentStageIndex]
                        charHrp.CFrame = CFrame.new(nextPoint)
                        print("📢 Sampai di Room " .. currentStageIndex)

                        -- SETELAH SAMPAI ROOM BARU -> MATIKAN UI
                        if roundUI then
                            roundUI.Visible = false
                            print("🙈 UI RoundCompleted Dimatikan Manual!")
                        end

                        task.wait(1)
                        isTransitioning = false
                    else
                        -- Room Akhir
                        local currentPoint = CHECKPOINTS[currentStageIndex]
                        charHrp.CFrame = CFrame.new(currentPoint)
                    end

                elseif not isTransitioning then
                    
                    -- ROUND MASIH JALAN: Bantai di room aktif
                    if humanoid then humanoid.AutoRotate = false end

                    local currentPoint = CHECKPOINTS[currentStageIndex]
                    charHrp.CFrame = CFrame.new(currentPoint) * CFrame.new(0, SAFE_OFFSET_Y, 0)
                    charHrp.AssemblyLinearVelocity = Vector3.zero
                    charHrp.AssemblyAngularVelocity = Vector3.zero

                end
            end)
        end
    end)

end)
